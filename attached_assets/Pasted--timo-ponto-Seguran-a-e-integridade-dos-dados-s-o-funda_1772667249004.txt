## 🛠️ **Código Seguro para Backup (Exemplo PHP)**

```php
<?php
class SistemaSeguro {
    private $db;
    private $logFile;
    
    function __construct() {
        $this->logFile = 'logs/seguranca_' . date('Y-m-d') . '.log';
        $this->conectarBanco();
    }
    
    // 📝 Log de segurança
    private function logSeguranca($mensagem, $tipo = 'INFO') {
        $log = date('Y-m-d H:i:s') . " [$tipo] $mensagem" . PHP_EOL;
        file_put_contents($this->logFile, $log, FILE_APPEND);
    }
    
    // 💾 Backup automático
    public function criarBackup() {
        try {
            $backupDir = 'backups/' . date('Y-m-d_H-i-s');
            mkdir($backupDir, 0755, true);
            
            // Backup do banco SQLite
            if (file_exists('database.sqlite')) {
                copy('database.sqlite', $backupDir . '/database.sqlite');
            }
            
            // Backup de arquivos importantes
            $arquivosImportantes = ['config.php', 'functions.php'];
            foreach ($arquivosImportantes as $arquivo) {
                if (file_exists($arquivo)) {
                    copy($arquivo, $backupDir . '/' . $arquivo);
                }
            }
            
            $this->logSeguranca("Backup criado em: $backupDir");
            return $backupDir;
            
        } catch (Exception $e) {
            $this->logSeguranca("ERRO no backup: " . $e->getMessage(), 'ERRO');
            return false;
        }
    }
    
    // ⏰ Função segura de data/hora
    public function getDataHoraSegura() {
        // Configurar fuso
        date_default_timezone_set('America/Sao_Paulo');
        
        // Tentar API externa (opcional, com timeout)
        $context = stream_context_create(['http' => ['timeout' => 2]]);
        
        try {
            $apiUrl = 'http://worldtimeapi.org/api/timezone/America/Sao_Paulo';
            $response = @file_get_contents($apiUrl, false, $context);
            
            if ($response) {
                $data = json_decode($response, true);
                if (isset($data['datetime'])) {
                    $this->logSeguranca("Data validada por API externa");
                    return date('Y-m-d H:i:s', strtotime($data['datetime']));
                }
            }
        } catch (Exception $e) {
            $this->logSeguranca("API falhou, usando local: " . $e->getMessage(), 'AVISO');
        }
        
        // Fallback seguro
        return date('Y-m-d H:i:s');
    }
    
    // 🔒 Função para registrar com segurança
    public function registrarComDataSegura($dados) {
        // Validar dados
        if (empty($dados)) {
            $this->logSeguranca("Tentativa de registro com dados vazios", 'AVISO');
            return false;
        }
        
        // Iniciar transaction
        $this->db->beginTransaction();
        
        try {
            // Obter data segura
            $dataAtual = $this->getDataHoraSegura();
            
            // Adicionar data aos dados
            $dados['data_registro'] = $dataAtual;
            $dados['data_original'] = $dataAtual; // Manter cópia
            
            // Log do registro
            $this->logSeguranca("Novo registro com data: $dataAtual");
            
            // Inserir no banco
            // ... seu código de inserção aqui ...
            
            // Commit
            $this->db->commit();
            return true;
            
        } catch (Exception $e) {
            // Rollback em caso de erro
            $this->db->rollBack();
            $this->logSeguranca("ERRO no registro: " . $e->getMessage(), 'ERRO');
            return false;
        }
    }
    
    // ↩️ Rollback de emergência
    public function rollbackEmergencia($backupDir) {
        if (file_exists($backupDir . '/database.sqlite')) {
            copy($backupDir . '/database.sqlite', 'database.sqlite');
            $this->logSeguranca("ROLLBACK executado do backup: $backupDir", 'ALERTA');
            return true;
        }
        return false;
    }
}

// Uso seguro
$sistema = new SistemaSeguro();

// Antes de qualquer alteração, fazer backup
$backup = $sistema->criarBackup();

// Agora sim, implementar as mudanças
// $sistema->registrarComDataSegura($_POST);

echo "✅ Sistema preparado! Backup criado em: $backup\n";
echo "⚠️  Verifique os logs em: logs/\n";
?>
```

    echo "✅ Data/Hora correta"
