2. Corrija o useFetchWithAuth para tratar falta de sessão
ts

import { createClient } from '@supabase/supabase-js';

const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

export const fetchWithAuth = async (url: string, options = {}) => {
  const {
    data: { session },
    error,
  } = await supabase.auth.getSession();

  if (!session || error) {
    console.error('[FetchWithAuth] Não foi possível obter token JWT para requisição:', url);
    throw new Error('Usuário não autenticado');
  }

  const token = session.access_token;

  const headers = {
    ...options?.headers,
    Authorization: `Bearer ${token}`,
    'Content-Type': 'application/json',
  };

  const res = await fetch(url, { ...options, headers });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`${res.status}: ${text}`);
  }

  return res.json();
};
3. Verifique se a rota realmente exige autenticação
Se /api/recebimentos/osasco_v2 for uma rota pública, remova a exigência de JWT.

Caso contrário, garanta que o token esteja presente no Authorization e que o backend esteja configurado para validá-lo.

✅ Dica extra
Se estiver usando Supabase, e o usuário está logado no app, o supabase.auth.getSession() deve retornar um token.

Mas se estiver testando a API em um ambiente separado (ex: Vercel + local), pode ocorrer falta de sessão.
