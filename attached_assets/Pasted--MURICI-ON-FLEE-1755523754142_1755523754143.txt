/*
================================================================================
1) SQL/01_backup_session.sql
- Cria um backup completo da tabela session ANTES de qualquer limpeza
- Seguro para rodar múltiplas vezes (usa tabela com carimbo de data)
================================================================================
*/
-- SQL/01_backup_session.sql
DO $$
DECLARE
  backup_table text := 'session_backup_' || to_char(now(), 'YYYYMMDD_HH24MI');
BEGIN
  EXECUTE format('CREATE TABLE %I AS TABLE session WITH DATA', backup_table);
  RAISE NOTICE 'Backup criado: %', backup_table;
END $$;

/*
================================================================================
2) SQL/02_cleanup_session_safe.sql
- Remove apenas sessões expiradas há mais de 30 dias
- Mantém as sessões recentes e válidas
- Executar após rodar o 01_backup_session.sql
================================================================================
*/
-- SQL/02_cleanup_session_safe.sql
BEGIN;
  -- Opcional: garantir que há índice em expire para acelerar o DELETE
  CREATE INDEX IF NOT EXISTS idx_session_expire ON session(expire);

  -- Remoção segura (apenas sessões antigas, já expiradas)
  DELETE FROM session
  WHERE expire < (NOW() - INTERVAL '30 days');

  -- Atualiza estatísticas para o planner do Postgres
  ANALYZE session;
COMMIT;

-- OBS: VACUUM é gerenciado pelo autovacuum do Supabase. Se necessário em janela de manutenção:
-- VACUUM (VERBOSE, ANALYZE) session;

-- SQL/04_views_metricas.sql
CREATE OR REPLACE VIEW v_abastecimentos_stats AS
SELECT
  current_date AS ref_date,
  (SELECT COUNT(*) FROM abastecimentos WHERE created_at >= now() - INTERVAL '24 hours') AS abastecimentos_24h,
  (SELECT COUNT(*) FROM abastecimentos WHERE created_at >= now() - INTERVAL '7 days') AS abastecimentos_7d,
  (SELECT COUNT(* ) FROM abastecimentos) AS abastecimentos_total;

CREATE OR REPLACE VIEW v_login_falhas_stats AS
SELECT
  current_date AS ref_date,
  (SELECT COUNT(*) FROM login_audit WHERE success = false AND created_at >= now() - INTERVAL '24 hours') AS falhas_24h,
  (SELECT COUNT(*) FROM login_audit WHERE created_at >= now() - INTERVAL '24 hours') AS tentativas_24h;

6) backend/src/jobs/sessionCleanup.ts
- Cron job diário para limpar sessões antigas, com logs e proteção por variável
================================================================================
*/
// backend/src/jobs/sessionCleanup.ts
import cron from 'node-cron';
import { Pool } from 'pg';
import pino from 'pino';

const logger = pino({ name: 'session-cleaner' });

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

export function startSessionCleanupJob() {
  const schedule = process.env.CRON_SESSION_CLEANUP || '0 3 * * *';
  const retentionDays = parseInt(process.env.SESSION_RETENTION_DAYS || '30', 10);

  cron.schedule(schedule, async () => {
    const client = await pool.connect();
    try {
      logger.info({ retentionDays }, 'Iniciando limpeza de sessions…');
      await client.query('BEGIN');
      await client.query('CREATE INDEX IF NOT EXISTS idx_session_expire ON session(expire)');
      const res = await client.query(
        `DELETE FROM session WHERE expire < (NOW() - INTERVAL '${retentionDays} days')`
      );
      await client.query('ANALYZE session');
      await client.query('COMMIT');
      logger.info({ deleted: res.rowCount }, 'Limpeza concluída');
    } catch (err) {
      await client.query('ROLLBACK');
      logger.error({ err }, 'Erro na limpeza de sessions');
    } finally {
      client.release();
    }
  }, {
    timezone: 'America/Sao_Paulo'
  });

  logger.info({ schedule }, 'Cron de limpeza de sessions agendado');
}

// backend/src/services/alerts.ts
import fetch from 'node-fetch';
import pino from 'pino';
const logger = pino({ name: 'alerts' });

export async function sendAlert(message: string) {
  const url = process.env.SLACK_WEBHOOK_URL;
  if (!url) {
    logger.warn('SLACK_WEBHOOK_URL não configurado. Alerta não enviado.');
    return;
  }
  await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ text: message })
  });
}

- Middleware para auditar logins e disparar alerta por falhas em massa
- Pressupõe uma tabela login_audit(email, base_id, success, ip, user_agent, created_at)

// backend/src/middleware/loginTracker.ts
import { Pool } from 'pg';
import type { Request, Response, NextFunction } from 'express';
import { sendAlert } from '../services/alerts';
import pino from 'pino';

const logger = pino({ name: 'login-tracker' });
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

export async function logLoginAttempt(req: Request, success: boolean, baseId?: string) {
  const client = await pool.connect();
  try {
    const { email } = req.body || {};
    const ip = (req.headers['x-forwarded-for'] as string) || req.ip || '';
    const ua = req.headers['user-agent'] || '';
    await client.query(
      `INSERT INTO login_audit(email, base_id, success, ip, user_agent, created_at)
       VALUES ($1, $2, $3, $4, $5, NOW())`,
      [email || null, baseId || null, success, ip, ua]
    );
  } finally {
    client.release();
  }
}

export async function failBurstGuard(req: Request, res: Response, next: NextFunction) {
  try {
    const threshold = parseInt(process.env.LOGIN_FAIL_ALERT_THRESHOLD || '15', 10);
    const windowMin = parseInt(process.env.LOGIN_FAIL_ALERT_WINDOW_MIN || '10', 10);

    const { rows } = await pool.query(
      `SELECT COUNT(*)::int AS n
         FROM login_audit
        WHERE success = false
          AND created_at >= NOW() - INTERVAL '${windowMin} minutes'`
    );

    const n = rows[0]?.n || 0;
    if (n >= threshold) {
      await sendAlert(`⚠️ Picos de falhas de login: ${n} nas últimas ${windowMin} min.`);
    }
  } catch (err) {
    logger.error({ err }, 'Erro no failBurstGuard');
  }
  next();
}

// backend/src/routes/metrics.ts
import { Router } from 'express';
import { Pool } from 'pg';

const router = Router();
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

router.get('/basic', async (_req, res) => {
  const client = await pool.connect();
  try {
    const q1 = client.query(`SELECT * FROM v_abastecimentos_stats`);
    const q2 = client.query(`SELECT * FROM v_login_falhas_stats`);
    const [r1, r2] = await Promise.all([q1, q2]);
    res.json({ abastecimentos: r1.rows[0], logins: r2.rows[0] });
  } catch (err: any) {
    res.status(500).json({ error: true, message: err.message });
  } finally {
    client.release();
  }
});

export default router;
