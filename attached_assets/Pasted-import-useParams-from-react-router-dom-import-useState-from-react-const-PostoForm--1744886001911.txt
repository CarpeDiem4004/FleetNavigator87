import { useParams } from 'react-router-dom';
import { useState } from 'react';

const PostoForm = () => {
  const { cidade } = useParams();
  const [form, setForm] = useState({
    placa: '',
    motorista: '',
    km: '',
    litros: '',
    tipo: 'diesel',
    movimento: 'entrada',
    destino: '',
  });

  const handleSubmit = (e) => {
    e.preventDefault();
    console.log('Enviando dados para Supabase:', form);
    alert('Abastecimento e movimentação registrados com sucesso!');
  };

  return (
    <div className="max-w-xl mx-auto mt-10 bg-white p-6 rounded shadow">
      <h2 className="text-xl font-bold mb-4 text-center">Posto {cidade.toUpperCase()}</h2>

      <form onSubmit={handleSubmit} className="grid grid-cols-1 gap-4">
        <input
          type="text"
          placeholder="Placa do Veículo"
          value={form.placa}
          onChange={(e) => setForm({ ...form, placa: e.target.value })}
          required
          className="p-2 border rounded"
        />
        <input
          type="text"
          placeholder="Nome do Motorista"
          value={form.motorista}
          onChange={(e) => setForm({ ...form, motorista: e.target.value })}
          required
          className="p-2 border rounded"
        />
        <input
          type="number"
          placeholder="KM Atual"
          value={form.km}
          onChange={(e) => setForm({ ...form, km: e.target.value })}
          required
          className="p-2 border rounded"
        />
        <input
          type="number"
          placeholder="Litros Abastecidos"
          value={form.litros}
          onChange={(e) => setForm({ ...form, litros: e.target.value })}
          required
          className="p-2 border rounded"
        />
        <select
          value={form.tipo}
          onChange={(e) => setForm({ ...form, tipo: e.target.value })}
          className="p-2 border rounded"
        >
          <option value="diesel">Diesel</option>
          <option value="arla">ARLA</option>
        </select>
        <select
          value={form.movimento}
          onChange={(e) => setForm({ ...form, movimento: e.target.value })}
          className="p-2 border rounded"
        >
          <option value="entrada">Entrada (pernoite)</option>
          <option value="saida_rota">Saída para Rota</option>
          <option value="saida_manutencao">Saída para Manutenção</option>
        </select>
        <button
          type="submit"
          className="bg-green-600 text-white p-2 rounded hover:bg-green-700"
        >
          Enviar Registro
        </button>
      </form>
    </div>
  );
};

export default PostoForm;
