import { useState } from "react";
import { createClient } from "@supabase/supabase-js";

const supabase = createClient("https://SEU-PROJETO.supabase.co", "CHAVE-ANON");

export default function AbastecimentoPostoRemedios() {
  const [form, setForm] = useState({
    placa: "",
    km: "",
    projeto: "",
    nome_motorista: "",
    rg_motorista: "",
    tipo_combustivel: "Diesel",
    valor_por_litro: "",
    litros_abastecidos: "",
    lavagem_frota: false,
  });

  const [status, setStatus] = useState("");

  const valorTotal = (form.valor_por_litro * form.litros_abastecidos).toFixed(2);

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    setForm({ ...form, [name]: type === "checkbox" ? checked : value });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    const { error } = await supabase.from("abastecimento_posto_remedios").insert([form]);
    if (error) {
      setStatus("Erro ao enviar: " + error.message);
    } else {
      setStatus("Registro enviado com sucesso!");
      setForm({
        placa: "",
        km: "",
        projeto: "",
        nome_motorista: "",
        rg_motorista: "",
        tipo_combustivel: "Diesel",
        valor_por_litro: "",
        litros_abastecidos: "",
        lavagem_frota: false,
      });
    }
  };

  return (
    <div className="p-4 max-w-xl mx-auto bg-white rounded-xl shadow-md space-y-4">
      <h2 className="text-xl font-bold">Abastecimento - Posto Remédios</h2>
      <form onSubmit={handleSubmit} className="space-y-3">
        <input name="placa" placeholder="Placa" value={form.placa} onChange={handleChange} required className="input" />
        <input name="km" type="number" placeholder="KM" value={form.km} onChange={handleChange} required className="input" />
        <input name="projeto" placeholder="Projeto" value={form.projeto} onChange={handleChange} required className="input" />
        <input name="nome_motorista" placeholder="Nome do Motorista" value={form.nome_motorista} onChange={handleChange} required className="input" />
        <input name="rg_motorista" placeholder="RG do Motorista" value={form.rg_motorista} onChange={handleChange} required className="input" />

        <select name="tipo_combustivel" value={form.tipo_combustivel} onChange={handleChange} className="input">
          <option>Diesel</option>
          <option>Gasolina</option>
          <option>Álcool</option>
        </select>

        <input name="valor_por_litro" type="number" step="0.01" placeholder="Valor por Litro" value={form.valor_por_litro} onChange={handleChange} required className="input" />
        <input name="litros_abastecidos" type="number" step="0.01" placeholder="Litros Abastecidos" value={form.litros_abastecidos} onChange={handleChange} required className="input" />

        <div className="text-gray-700">Valor Total: R$ {isNaN(valorTotal) ? "0.00" : valorTotal}</div>

        <label className="flex items-center gap-2">
          <input type="checkbox" name="lavagem_frota" checked={form.lavagem_frota} onChange={handleChange} />
          Lavagem de Frota
        </label>

        <button type="submit" className="btn bg-green-600 text-white w-full">Enviar</button>
        {status && <p className="text-sm text-center mt-2">{status}</p>}
      </form>
    </div>
  );
}
