// js/auth.js
function entrar() {
  const login = document.getElementById("login").value;
  const senha = document.getElementById("senha").value;

  const user = usuarios.find(u => u.login === login && u.senha === senha);

  if (!user) {
    alert("Login inválido");
    return;
  }

  localStorage.setItem("usuario", JSON.stringify(user));

  if (user.role === "BASE") location.href = "base.html";
  else location.href = "admin.html";
}

// js/base.js
const usuario = JSON.parse(localStorage.getItem("usuario"));
document.getElementById("titulo").innerText = `Base ${usuario.base}`;

const lista = document.getElementById("lista");
const frota = veiculos.filter(v => v.base === usuario.base);

frota.forEach(v => {
  lista.innerHTML += `
    <tr>
      <td>${v.placa}</td>
      <td>
        <select>
          <option>DISPONIVEL</option>
          <option>ROTA</option>
          <option>MANUTENCAO</option>
          <option>FALTA_EQUIPE</option>
        </select>
      </td>
      <td><input placeholder="Oficina"></td>
      <td><input type="date"></td>
    </tr>
  `;
});

function salvar() {
  atualizacoes.push({
    base: usuario.base,
    data: new Date().toISOString().split("T")[0]
  });
  alert("Atualização diária salva!");
}

// js/admin.js
const hoje = new Date().toISOString().split("T")[0];
const tbody = document.getElementById("bases");

const bases = [...new Set(usuarios.filter(u => u.role === "BASE").map(u => u.base))];

bases.forEach(base => {
  const atualizou = atualizacoes.some(a => a.base === base && a.data === hoje);
  tbody.innerHTML += `
    <tr style="background:${atualizou ? '#cfc' : '#fcc'}">
      <td>${base}</td>
      <td>${atualizou ? "OK" : "PENDENTE"}</td>
    </tr>
  `;
});
