## 1. Configuração do Banco de Dados (schema.sql):

```sql
-- Tabela de usuários
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    role ENUM('admin', 'operador_frota', 'visualizador') DEFAULT 'visualizador',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabela de permissões por card
CREATE TABLE card_permissions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    card_id VARCHAR(50) NOT NULL,
    can_view BOOLEAN DEFAULT true,
    can_access BOOLEAN DEFAULT true,
    FOREIGN KEY (user_id) REFERENCES users(id),
    UNIQUE KEY unique_user_card (user_id, card_id)
);

-- Cards disponíveis no sistema
CREATE TABLE cards (
    id VARCHAR(50) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    default_for_operator BOOLEAN DEFAULT false
);
```

## 2. Backend - Model de Permissões (Permission.js):

```javascript
// backend/models/Permission.js
const db = require('../config/database');

class Permission {
    static async getUserPermissions(userId) {
        const [rows] = await db.query(`
            SELECT c.*, 
                   COALESCE(cp.can_view, c.default_for_operator) as can_view,
                   COALESCE(cp.can_access, c.default_for_operator) as can_access
            FROM cards c
            LEFT JOIN card_permissions cp ON c.id = cp.card_id AND cp.user_id = ?
            ORDER BY c.id
        `, [userId]);
        
        return rows;
    }

    static async updateUserCardPermission(userId, cardId, permissions) {
        const { can_view, can_access } = permissions;
        
        await db.query(`
            INSERT INTO card_permissions (user_id, card_id, can_view, can_access)
            VALUES (?, ?, ?, ?)
            ON DUPLICATE KEY UPDATE
            can_view = VALUES(can_view),
            can_access = VALUES(can_access)
        `, [userId, cardId, can_view, can_access]);
    }
}

module.exports = Permission;
```

## 3. Componente de Card com Controle de Acesso (CardWrapper.jsx):

```jsx
// frontend/src/components/cards/CardWrapper.jsx
import React from 'react';
import { useAuth } from '../../contexts/AuthContext';

const CardWrapper = ({ cardId, children, title, action }) => {
  const { userPermissions } = useAuth();
  
  const permission = userPermissions?.find(p => p.id === cardId);
  
  if (!permission?.can_view) {
    return null; // Não renderiza o card
  }

  const handleClick = () => {
    if (permission?.can_access && action) {
      action();
    } else if (!permission?.can_access) {
      alert('Você não tem permissão para acessar este módulo');
    }
  };

  return (
    <div 
      className={`card ${!permission?.can_access ? 'card-disabled' : ''}`}
      onClick={handleClick}
      style={{
        opacity: permission?.can_access ? 1 : 0.7,
        cursor: permission?.can_access ? 'pointer' : 'not-allowed',
        border: permission?.can_access ? 'none' : '1px dashed #ccc'
      }}
    >
      <div className="card-header">
        <h3>{title}</h3>
        {permission?.can_access ? (
          <span className="access-badge">Acessar →</span>
        ) : (
          <span className="access-badge disabled">Acesso restrito</span>
        )}
      </div>
      {children}
    </div>
  );
};

export default CardWrapper;
```

## 4. Contexto de Autenticação (AuthContext.jsx):

```jsx
// frontend/src/contexts/AuthContext.jsx
import React, { createContext, useState, useContext, useEffect } from 'react';
import api from '../services/api';

const AuthContext = createContext();

export const useAuth = () => useContext(AuthContext);

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [userPermissions, setUserPermissions] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadUser();
  }, []);

  const loadUser = async () => {
    const token = localStorage.getItem('token');
    if (token) {
      try {
        const response = await api.get('/auth/me');
        setUser(response.data.user);
        await loadPermissions(response.data.user.id);
      } catch (error) {
        console.error('Erro ao carregar usuário:', error);
      }
    }
    setLoading(false);
  };

  const loadPermissions = async (userId) => {
    try {
      const response = await api.get(`/permissions/user/${userId}`);
      setUserPermissions(response.data);
    } catch (error) {
      console.error('Erro ao carregar permissões:', error);
    }
  };

  const login = async (credentials) => {
    const response = await api.post('/auth/login', credentials);
    localStorage.setItem('token', response.data.token);
    setUser(response.data.user);
    await loadPermissions(response.data.user.id);
    return response.data;
  };

  const logout = () => {
    localStorage.removeItem('token');
    setUser(null);
    setUserPermissions([]);
  };

  return (
    <AuthContext.Provider value={{ 
      user, 
      userPermissions, 
      login, 
      logout, 
      loading,
      loadPermissions 
    }}>
      {children}
    </AuthContext.Provider>
  );
};
```

## 6. Painel Administrativo para Gerenciar Permissões:

```jsx
// frontend/src/components/admin/PermissionManager.jsx
import React, { useState, useEffect } from 'react';
import api from '../../services/api';

const PermissionManager = () => {
  const [users, setUsers] = useState([]);
  const [cards, setCards] = useState([]);
  const [selectedUser, setSelectedUser] = useState(null);
  const [permissions, setPermissions] = useState({});

  useEffect(() => {
    loadUsers();
    loadCards();
  }, []);

  const loadUsers = async () => {
    const response = await api.get('/users');
    setUsers(response.data);
  };

  const loadCards = async () => {
    const response = await api.get('/cards');
    setCards(response.data);
  };

  const loadUserPermissions = async (userId) => {
    const response = await api.get(`/permissions/user/${userId}`);
    const perms = {};
    response.data.forEach(p => {
      perms[p.id] = { view: p.can_view, access: p.can_access };
    });
    setPermissions(perms);
  };

  const handleUserSelect = (userId) => {
    setSelectedUser(userId);
    loadUserPermissions(userId);
  };

  const updatePermission = async (cardId, type, value) => {
    const updatedPerms = {
      ...permissions[cardId],
      [type]: value
    };

    await api.post(`/permissions/update`, {
      userId: selectedUser,
      cardId,
      permissions: {
        can_view: updatedPerms.view,
        can_access: updatedPerms.access
      }
    });

    setPermissions({
      ...permissions,
      [cardId]: updatedPerms
    });
  };

  return (
    <div className="permission-manager">
      <h2>Gerenciar Permissões de Cards</h2>
      
      <div className="user-selector">
        <label>Selecione o Operador:</label>
        <select onChange={(e) => handleUserSelect(e.target.value)}>
          <option value="">Selecione...</option>
          {users.filter(u => u.role === 'operador_frota').map(user => (
            <option key={user.id} value={user.id}>
              {user.username} - {user.email}
            </option>
          ))}
        </select>
      </div>

      {selectedUser && (
        <div className="permissions-grid">
          <table>
            <thead>
              <tr>
                <th>Card</th>
                <th>Visualizar</th>
                <th>Acessar</th>
              </tr>
            </thead>
            <tbody>
              {cards.map(card => (
                <tr key={card.id}>
                  <td>{card.name}</td>
                  <td>
                    <input
                      type="checkbox"
                      checked={permissions[card.id]?.view || false}
                      onChange={(e) => updatePermission(card.id, 'view', e.target.checked)}
                    />
                  </td>
                  <td>
                    <input
                      type="checkbox"
                      checked={permissions[card.id]?.access || false}
                      onChange={(e) => updatePermission(card.id, 'access', e.target.checked)}
                      disabled={!permissions[card.id]?.view}
                    />
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}
    </div>
  );
};

export default PermissionManager;
```

## 8. Como Implementar no Replit:

1. **Crie as pastas** conforme a estrutura acima
2. **Instale as dependências**:
```bash
npm install express react react-dom mysql2 jsonwebtoken bcrypt
npm install -D @types/react
```

3. **Configure as variáveis de ambiente** no arquivo `.env`:
```env
DB_HOST=seu-host
DB_USER=seu-usuario
DB_PASSWORD=sua-senha
DB_NAME=gestao_frota
JWT_SECRET=seu-segredo-jwt
```

4. **Execute o script SQL** para criar as tabelas

5. **Crie um usuário administrador** manualmente no banco de dados

## 9. Fluxo de Uso:

1. **Admin cria usuário** com role 'operador_frota'
2. **Admin acessa o PermissionManager** e configura quais cards cada operador pode ver/acessar
3. **Operador faz login** e vê apenas os cards liberados
4. **Cards sem permissão de visualização** não aparecem
5. **Cards com permissão de visualização mas sem acesso** aparecem "acinzentados" e não podem ser clicados

Esta estrutura te dá controle total sobre o que cada operador pode ver e acessar no sistema de gestão de frota, diretamente do Replit!
