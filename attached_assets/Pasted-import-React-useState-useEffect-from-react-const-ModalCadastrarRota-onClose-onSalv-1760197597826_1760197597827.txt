import React, { useState, useEffect } from "react";

const ModalCadastrarRota = ({ onClose, onSalvar }) => {
  const [origem, setOrigem] = useState("");
  const [destino, setDestino] = useState("");
  const [distancia, setDistancia] = useState("");
  const [observacoes, setObservacoes] = useState("");
  const [statusConsulta, setStatusConsulta] = useState("");

  // Substitua pela sua API Key do Google Maps
  const GOOGLE_API_KEY = "SUA_API_KEY_GOOGLE_MAPS";

  useEffect(() => {
    const calcularDistancia = async () => {
      if (origem.trim() && destino.trim()) {
        try {
          setStatusConsulta("Consultando distância...");
          const response = await fetch(
            `https://maps.googleapis.com/maps/api/distancematrix/json?origins=${encodeURIComponent(
              origem
            )}&destinations=${encodeURIComponent(
              destino
            )}&units=metric&key=${GOOGLE_API_KEY}`
          );
          const data = await response.json();

          if (data.rows?.[0]?.elements?.[0]?.status === "OK") {
            const distanciaKm = data.rows[0].elements[0].distance.value / 1000;
            setDistancia(distanciaKm.toFixed(0));
            setStatusConsulta(`Distância calculada: ${distanciaKm.toFixed(0)} km`);
          } else {
            setStatusConsulta("Não foi possível calcular a distância.");
          }
        } catch (error) {
          console.error("Erro ao consultar Google Maps:", error);
          setStatusConsulta("Erro ao consultar distância.");
        }
      }
    };

    calcularDistancia();
  }, [origem, destino]);

  const handleSubmit = (e) => {
    e.preventDefault();
    onSalvar({ origem, destino, distancia, observacoes });
  };

  return (
    <div className="fixed inset-0 flex items-center justify-center bg-black/40 z-50">
      <div className="bg-blue-100 rounded-2xl shadow-lg w-[500px] p-6">
        <h2 className="text-green-700 font-semibold text-lg mb-1 flex items-center gap-2">
          <span className="text-xl">+</span> Cadastrar Nova Rota
        </h2>
        <p className="text-sm text-gray-700 mb-4">
          Preencha os dados da nova rota Line Haul
        </p>

        <form onSubmit={handleSubmit} className="space-y-4">
          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="text-sm font-medium text-gray-700">
                Ponto de Origem *
              </label>
              <input
                type="text"
                placeholder="Ex: Rio de Janeiro"
                value={origem}
                onChange={(e) => setOrigem(e.target.value)}
                className="w-full border rounded-lg p-2 outline-none"
                required
              />
            </div>

            <div>
              <label className="text-sm font-medium text-gray-700">
                Ponto de Destino *
              </label>
              <input
                type="text"
                placeholder="Ex: São Paulo"
                value={destino}
                onChange={(e) => setDestino(e.target.value)}
                className="w-full border rounded-lg p-2 outline-none"
                required
              />
            </div>
          </div>

          <div>
            <label className="text-sm font-medium text-gray-700">
              Distância Total (km) *
            </label>
            <input
              type="text"
              placeholder="Ex: 450"
              value={distancia}
              onChange={(e) => setDistancia(e.target.value)}
              className="w-full border rounded-lg p-2 outline-none bg-white"
              required
              readOnly
            />
            {statusConsulta && (
              <p className="text-xs text-gray-600 mt-1">{statusConsulta}</p>
            )}
          </div>

          <div>
            <label className="text-sm font-medium text-gray-700">
              Observações
            </label>
            <textarea
              placeholder="Informações adicionais sobre a rota..."
              value={observacoes}
              onChange={(e) => setObservacoes(e.target.value)}
              className="w-full border rounded-lg p-2 outline-none"
              rows={3}
            />
          </div>

          <div className="flex justify-end gap-3 mt-4">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 bg-gray-200 rounded-lg hover:bg-gray-300 transition"
            >
              Cancelar
            </button>
            <button
              type="submit"
              className="px-4 py-2 bg-green-400 text-white rounded-lg hover:bg-green-500 transition"
            >
              + Cadastrar Rota
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};

export default ModalCadastrarRota;
