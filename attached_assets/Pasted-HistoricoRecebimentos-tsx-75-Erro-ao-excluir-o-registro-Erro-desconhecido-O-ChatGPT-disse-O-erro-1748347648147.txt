import axios from 'axios';
import { supabase } from '../lib/supabaseClient';

const excluirRegistro = async (id: string) => {
  try {
    const { data: { session } } = await supabase.auth.getSession();

    const token = session?.access_token;
    if (!token) throw new Error("Usuário não autenticado");

    await axios.delete(`${API_URL}/recebimentos/${id}`, {
      headers: {
        Authorization: `Bearer ${token}`
      }
    });

    alert("Registro excluído com sucesso!");
  } catch (error: any) {
    console.error("Erro ao excluir o registro:", error);
    if (error.response) {
      alert(`Erro ao excluir: ${error.response.status} - ${error.response.data.message || error.response.statusText}`);
    } else if (error.request) {
      alert("Erro ao excluir: Sem resposta do servidor");
    } else {
      alert(`Erro ao excluir: ${error.message}`);
    }
  }
};
