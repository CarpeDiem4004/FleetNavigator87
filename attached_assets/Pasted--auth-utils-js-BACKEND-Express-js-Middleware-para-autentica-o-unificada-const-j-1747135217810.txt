// ==== BACKEND: Express.js Middleware para autenticação unificada ====
const jwt = require("jsonwebtoken");
const supabase = require("./utils/supabaseClient"); // Ajuste conforme seu projeto

async function unifiedAuthMiddleware(req, res, next) {
  try {
    let token = null;

    // Busca token em diferentes locais
    if (req.headers.authorization?.startsWith("Bearer ")) {
      token = req.headers.authorization.split(" ")[1];
    } else if (req.session?.user?.token) {
      token = req.session.user.token;
    } else if (req.cookies?.authToken) {
      token = req.cookies.authToken;
    }

    if (!token) throw new Error("Token não fornecido");

    // Tenta verificar como JWT local
    try {
      const decoded = jwt.verify(token, process.env.JWT_SECRET);
      req.user = decoded;
      return next();
    } catch (jwtErr) {
      // Se falhar, tenta via Supabase
      const { data, error } = await supabase.auth.getUser(token);
      if (error || !data?.user) throw new Error("Token inválido Supabase");
      req.user = data.user;
      return next();
    }
  } catch (err) {
    return res.status(401).json({ error: "Não autorizado", detail: err.message });
  }
}

// ==== BACKEND: Validação de permissões ====
function hasRole(user, allowedRoles = []) {
  return allowedRoles.includes(user.role);
}

// ==== FRONTEND: Token storage utilitário ====
export function saveAuthToken(token) {
  localStorage.setItem("authToken", token);
}

export function getAuthToken() {
  return localStorage.getItem("authToken");
}

export function clearAuthToken() {
  localStorage.removeItem("authToken");
}

// ==== FRONTEND: Hook para fetch com autenticação (React) ====
export function useFetchWithAuth() {
  const fetchWithAuth = async (url, options = {}) => {
    const token = getAuthToken();

    const headers = {
      ...(options.headers || {}),
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    };

    const response = await fetch(url, {
      ...options,
      headers,
    });

    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}));
      throw new Error(errorData?.error || "Erro na requisição");
    }

    return response.json();
  };

  return fetchWithAuth;
}

// ==== Exportações para uso no backend ====
module.exports = {
  unifiedAuthMiddleware,
  hasRole
};
